# bit_stuffing.py
def stuff_bits(data):
    cnt, s = 0, []
    for ch in data:
        if ch == '1':
            cnt += 1
            s.append(ch)
            if cnt == 5:
                s.append('0')
                cnt = 0
        else:
            s.append(ch)
            cnt = 0
    return "01111110" + "".join(s) + "01111110"

def unstuff_bits(s):
    cnt, unstuffed, i = 0, [], 8
    while i < len(s) - 8:
        ch = s[i]
        if ch == '1':
            cnt += 1
            unstuffed.append(ch)
            if cnt == 5:
                i += 1  # Skip stuffed zero
                cnt = 0
        else:
            unstuffed.append(ch)
            cnt = 0
        i += 1
    return "".join(unstuffed)

if __name__ == '__main__':
    data = input("Enter bit string: ")
    stuffed = stuff_bits(data)
    print("Stuffed data:  ", stuffed)
    unstuffed = unstuff_bits(stuffed)
    print("Unstuffed data:", unstuffed)
