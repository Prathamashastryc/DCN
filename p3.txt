def xor(a, b):
    return "".join('0' if a[i] == b[i] else '1' for i in range(1, len(b)))

def divide(dividend, divisor):
    n = len(divisor)
    temp = dividend[:n]
    for i in range(n, len(dividend) + 1):
        temp = xor(temp, divisor) if temp[0] == '1' else xor(temp, '0' * n)
        if i < len(dividend):
            temp += dividend[i]
    return temp

if __name__ == "__main__":
    data = input("Data: ")
    divisor = input("Generator: ")
    remainder = divide(data + '0' * (len(divisor) - 1), divisor)
    print("Remainder:", remainder)
    print("Codeword:", data + remainder)
